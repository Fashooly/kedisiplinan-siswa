/** * SISTEM KEDISIPLINAN & MUTABA'AH MURID - SDIT AZ ZAHRA * Backend Google Apps Script (Code.gs) */ /** * Melayani antarmuka HTML aplikasi web ATAU REST API JSON untuk Vercel / GitHub */ function doGet(e) { if (e && e.parameter && (e.parameter.action || e.parameter.api === 'true')) { const action = e.parameter.action || 'getInitialData'; let result = { status: 'error', message: 'Aksi tidak dikenali' }; if (action === 'getInitialData') { result = getInitialData(); } else if (action === 'ping') { result = { status: 'success', message: 'API SDIT Az Zahra aktif!' }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } try { return HtmlService.createHtmlOutputFromFile('index') .setTitle('Sistem Kedisiplinan SDIT Az Zahra') .addMetaTag('viewport', 'width=device-width, initial-scale=1.0, maximum-scale=5.0, viewport-fit=cover') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } catch (err) { return HtmlService.createTemplateFromFile('index') .evaluate() .setTitle('Sistem Kedisiplinan SDIT Az Zahra') .addMetaTag('viewport', 'width=device-width, initial-scale=1.0, maximum-scale=5.0, viewport-fit=cover') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } } /** * Endpoint POST untuk menerima data dari Vercel / Frontend Eksternal */ function doPost(e) { const lock = LockService.getScriptLock(); try { // Kunci proses selama 15 detik agar antrean data tidak bentrok lock.waitLock(15000); let payload = {}; if (e && e.postData && e.postData.contents) { payload = JSON.parse(e.postData.contents); } const action = (e && e.parameter && e.parameter.action) || payload.action; let result = { status: 'error', message: 'Aksi POST tidak dikenali' }; if (action === 'simpanCatatan') { result = simpanCatatan(payload.data || payload); } else if (action === 'tambahSantriBaru') { result = tambahSantriBaru(payload.data || payload); } else if (action === 'tambahSantriBatch') { result = tambahSantriBatch(payload.data || payload.list || []); } else if (action === 'updateStatusSantri') { result = updateStatusSantri(payload.data?.nis || payload.nis, payload.data || payload); } else if (action === 'batchUpdateSiswa') { result = batchUpdateSiswa(payload.updates || payload.data || []); } SpreadsheetApp.flush(); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Gagal memproses di server: ' + err.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } /** * Memastikan sheet DataSiswa selalu siap pakai (Auto-Create jika hilang) */ function pastikanSheetSiswa(ss) { let sheetSiswa = ss.getSheetByName('DataSiswa'); if (!sheetSiswa) { sheetSiswa = ss.insertSheet('DataSiswa'); sheetSiswa.appendRow([ 'NIS', 'Nama Lengkap', 'Kelas', 'Poin Saat Ini', 'Nama Wali', 'No HP Wali', 'Status', 'Catatan Khusus', 'Terakhir Diperbarui', 'Nama Wali Kelas', 'No WA Wali Kelas' ]); sheetSiswa.getRange('A1:K1').setBackground('#1b4332').setFontColor('#ffffff').setFontWeight('bold'); } return sheetSiswa; } /** * Memastikan sheet RiwayatCatatan selalu siap pakai */ function pastikanSheetRiwayat(ss) { let sheetRiwayat = ss.getSheetByName('RiwayatCatatan'); if (!sheetRiwayat) { sheetRiwayat = ss.insertSheet('RiwayatCatatan'); sheetRiwayat.appendRow([ 'ID Catatan', 'Tanggal & Waktu', 'NIS', 'Nama Murid', 'Kelas', 'Tipe', 'Judul Catatan', 'Perubahan Poin', 'Saldo Akhir', 'Tindakan / Pembinaan', 'Pencatat (Bapak/Ibu Walas)' ]); sheetRiwayat.getRange('A1:K1').setBackground('#2d6a4f').setFontColor('#ffffff').setFontWeight('bold'); } return sheetRiwayat; } /** * Inisialisasi struktur sheet otomatis jika belum tersedia */ function setupDatabaseAwal() { const ss = SpreadsheetApp.getActiveSpreadsheet(); pastikanSheetSiswa(ss); pastikanSheetRiwayat(ss); let sheetPengaturan = ss.getSheetByName('KatalogKaidah'); if (!sheetPengaturan) { sheetPengaturan = ss.insertSheet('KatalogKaidah'); sheetPengaturan.appendRow(['Tipe', 'Kategori', 'Deskripsi Aturan', 'Poin', 'Konsekuensi Default']); sheetPengaturan.getRange('A1:E1').setBackground('#40916c').setFontColor('#ffffff').setFontWeight('bold'); } return { status: 'success', message: 'Database SDIT Az Zahra berhasil diinisialisasi!' }; } /** * Mengambil seluruh data murid dan riwayat catatan */ function getInitialData() { const cache = CacheService.getScriptCache(); const cachedData = cache.get('initial_data_azzahra_v2'); if (cachedData) { try { return JSON.parse(cachedData); } catch (e) {} } try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetSiswa = pastikanSheetSiswa(ss); let siswaList = []; const dataSiswa = sheetSiswa.getDataRange().getValues(); for (let i = 1; i < dataSiswa.length; i++) { if (!dataSiswa[i][0]) continue; siswaList.push({ nis: String(dataSiswa[i][0]), nama: String(dataSiswa[i][1]), kelas: String(dataSiswa[i][2]), poin: Number(dataSiswa[i][3]) || 100, wali: String(dataSiswa[i][4] || ''), hpWali: String(dataSiswa[i][5] || ''), status: String(dataSiswa[i][6] || 'Aktif'), catatan: String(dataSiswa[i][7] || '-'), waliKelas: String(dataSiswa[i][9] || ''), hpWaliKelas: String(dataSiswa[i][10] || '') }); } const sheetRiwayat = pastikanSheetRiwayat(ss); let riwayatList = []; const lastRow = sheetRiwayat.getLastRow(); if (lastRow > 1) { const startRow = Math.max(2, lastRow - 99); const numRows = lastRow - startRow + 1; const dataRiwayat = sheetRiwayat.getRange(startRow, 1, numRows, 11).getValues(); for (let i = 0; i < dataRiwayat.length; i++) { if (!dataRiwayat[i][0]) continue; riwayatList.push({ id: String(dataRiwayat[i][0]), tanggal: dataRiwayat[i][1] instanceof Date ? Utilities.formatDate(dataRiwayat[i][1], 'Asia/Jakarta', 'yyyy-MM-dd HH:mm') : String(dataRiwayat[i][1]), nis: String(dataRiwayat[i][2]), nama: String(dataRiwayat[i][3]), kelas: String(dataRiwayat[i][4]), tipe: String(dataRiwayat[i][5]), judul: String(dataRiwayat[i][6]), poin: Number(dataRiwayat[i][7]), saldoAkhir: Number(dataRiwayat[i][8]), tindakan: String(dataRiwayat[i][9] || '-'), guru: String(dataRiwayat[i][10] || 'Tim Kesiswaan') }); } } const responsePayload = { status: 'success', data: { siswa: siswaList, riwayat: riwayatList.reverse(), kaidah: [] } }; try { cache.put('initial_data_azzahra_v2', JSON.stringify(responsePayload), 180); } catch (cacheErr) {} return responsePayload; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Menyimpan catatan pelanggaran atau kebaikan baru & memperbarui saldo poin murid */ function simpanCatatan(catatanData) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetSiswa = pastikanSheetSiswa(ss); const sheetRiwayat = pastikanSheetRiwayat(ss); const dataSiswa = sheetSiswa.getDataRange().getValues(); let targetRow = -1; let saldoPoinSebelumnya = 100; let namaSiswa = catatanData.nama; let kelasSiswa = catatanData.kelas; for (let i = 1; i < dataSiswa.length; i++) { if (String(dataSiswa[i][0]).trim() === String(catatanData.nis).trim()) { targetRow = i + 1; saldoPoinSebelumnya = Number(dataSiswa[i][3]) || 100; namaSiswa = String(dataSiswa[i][1]); kelasSiswa = String(dataSiswa[i][2]); break; } } if (targetRow === -1) { throw new Error('Murid dengan NIS ' + catatanData.nis + ' tidak ditemukan di database.'); } const perubahanPoin = Number(catatanData.poin); const saldoAkhir = Math.max(0, saldoPoinSebelumnya + perubahanPoin); sheetSiswa.getRange(targetRow, 4).setValue(saldoAkhir); sheetSiswa.getRange(targetRow, 9).setValue(new Date()); const recordId = 'REC-' + new Date().getTime(); sheetRiwayat.appendRow([ recordId, new Date(), catatanData.nis, namaSiswa, kelasSiswa, catatanData.tipe, catatanData.judul, perubahanPoin, saldoAkhir, catatanData.tindakan || '-', catatanData.guru || 'Tim Kesiswaan' ]); SpreadsheetApp.flush(); return { status: 'success', message: 'Catatan ' + catatanData.tipe.toLowerCase() + ' berhasil dicatat untuk ' + namaSiswa + '.', saldoAkhir: saldoAkhir, recordId: recordId }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Mendaftarkan murid baru secara perorangan */ function tambahSantriBaru(santriData) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = pastikanSheetSiswa(ss); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(santriData.nis).trim()) { return { status: 'error', message: 'NIS ' + santriData.nis + ' sudah terdaftar.' }; } } sheet.appendRow([ String(santriData.nis).trim(), String(santriData.nama).trim(), String(santriData.kelas).trim(), 100, String(santriData.wali || '').trim(), String(santriData.hpWali || '').trim(), 'Aktif', String(santriData.catatan || '-').trim(), new Date(), String(santriData.waliKelas || '').trim(), String(santriData.hpWaliKelas || '').trim() ]); SpreadsheetApp.flush(); return { status: 'success', message: 'Murid ' + santriData.nama + ' berhasil ditambahkan ke Google Sheets.' }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Mendaftarkan murid secara massal dari impor berkas Excel */ function tambahSantriBatch(santriList) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = pastikanSheetSiswa(ss); const data = sheet.getDataRange().getValues(); const existingNisSet = new Set(); for (let i = 1; i < data.length; i++) { if (data[i][0]) existingNisSet.add(String(data[i][0]).trim()); } const now = new Date(); const rowsToAppend = []; let duplikatCount = 0; santriList.forEach(s => { const nisClean = String(s.nis).trim(); if (!nisClean) return; if (existingNisSet.has(nisClean)) { duplikatCount++; return; } existingNisSet.add(nisClean); rowsToAppend.push([ nisClean, String(s.nama).trim(), String(s.kelas || '1 Umar bin Abdul Aziz').trim(), Number(s.poin) || 100, String(s.wali || '').trim(), String(s.hpWali || '').trim(), 'Aktif', String(s.catatan || '-').trim(), now, String(s.waliKelas || '').trim(), String(s.hpWaliKelas || '').trim() ]); }); if (rowsToAppend.length > 0) { const startRow = sheet.getLastRow() + 1; sheet.getRange(startRow, 1, rowsToAppend.length, 11).setValues(rowsToAppend); SpreadsheetApp.flush(); } return { status: 'success', insertedCount: rowsToAppend.length, duplikatCount: duplikatCount, message: 'Alhamdulillah, ' + rowsToAppend.length + ' murid berhasil diimpor ke Google Sheets.' }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Memperbarui status individual murid */ function updateStatusSantri(nis, statusData) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = pastikanSheetSiswa(ss); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(nis).trim()) { const statusBaru = statusData.status?.status || statusData.status; if (statusBaru) sheet.getRange(i + 1, 7).setValue(statusBaru); if (statusData.kelas) sheet.getRange(i + 1, 3).setValue(statusData.kelas); if (typeof statusData.poin !== 'undefined') sheet.getRange(i + 1, 4).setValue(statusData.poin); sheet.getRange(i + 1, 9).setValue(new Date()); SpreadsheetApp.flush(); return { status: 'success', message: 'Status murid berhasil diperbarui di Google Sheets.' }; } } return { status: 'error', message: 'NIS tidak ditemukan di Google Sheets.' }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Pembaruan massal kenaikan kelas */ function batchUpdateSiswa(updatesList) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = pastikanSheetSiswa(ss); const data = sheet.getDataRange().getValues(); const now = new Date(); const lookupMap = {}; updatesList.forEach(u => { lookupMap[String(u.nis).trim()] = u; }); let updatedCount = 0; for (let i = 1; i < data.length; i++) { const rowNis = String(data[i][0]).trim(); if (lookupMap[rowNis]) { const item = lookupMap[rowNis]; if (item.kelas) sheet.getRange(i + 1, 3).setValue(item.kelas); if (typeof item.poin !== 'undefined') sheet.getRange(i + 1, 4).setValue(item.poin); if (item.status) sheet.getRange(i + 1, 7).setValue(item.status); if (item.waliKelas) sheet.getRange(i + 1, 10).setValue(item.waliKelas); if (item.hpWaliKelas) sheet.getRange(i + 1, 11).setValue(item.hpWaliKelas); sheet.getRange(i + 1, 9).setValue(now); updatedCount++; } } SpreadsheetApp.flush(); return { status: 'success', message: 'Alhamdulillah, ' + updatedCount + ' data murid berhasil diperbarui di Google Sheets.' }; } catch (error) { return { status: 'error', message: error.toString() }; } }
