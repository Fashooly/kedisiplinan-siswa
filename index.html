} else if (action === 'tambahSantriBaru') { result = tambahSantriBaru(payload.data || payload); } else if (action === 'tambahSantriBatch') { result = tambahSantriBatch(payload.data || payload.list || []); } else if (action === 'updateStatusSantri') { result = updateStatusSantri(payload.nis, payload.data || payload); } else if (action === 'batchUpdateSiswa') { result = batchUpdateSiswa(payload.updates || payload.data || []); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Gagal memproses request: ' + err.toString() })).setMimeType(ContentService.MimeType.JSON); Baca Katalog Kaidah const sheetKaidah = ss.getSheetByName('KatalogKaidah'); let kaidahList = []; if (sheetKaidah) { const dataKaidah = sheetKaidah.getDataRange().getValues(); for (let i = 1; i < dataKaidah.length; i++) { if (!dataKaidah[i][2]) continue; kaidahList.push({ tipe: String(dataKaidah[i][0]), kategori: String(dataKaidah[i][1]), judul: String(dataKaidah[i][2]), poin: Number(dataKaidah[i][3]), konsekuensi: String(dataKaidah[i][4] || '') }); } } const responsePayload = { status: 'success', data: { siswa: siswaList, riwayat: riwayatList.reverse(), kaidah: kaidahList } }; // Simpan ke Cache RAM Google selama 10 menit (agar request berikutnya < 0.3 detik) try { cache.put('initial_data_azzahra_v2', JSON.stringify(responsePayload), 600); } catch (cacheErr) { // Abaikan jika ukuran melampaui limit } return responsePayload; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Menyimpan catatan pelanggaran atau kebaikan baru & memperbarui saldo poin murid */ function simpanCatatan(catatanData) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetSiswa = ss.getSheetByName('DataSiswa'); const sheetRiwayat = ss.getSheetByName('RiwayatCatatan'); if (!sheetSiswa || !sheetRiwayat) { throw new Error('Sheet DataSiswa atau RiwayatCatatan belum dibuat.'); } const dataSiswa = sheetSiswa.getDataRange().getValues(); let targetRow = -1; let saldoPoinSebelumnya = 100; let namaSiswa = catatanData.nama; let kelasSiswa = catatanData.kelas; for (let i = 1; i < dataSiswa.length; i++) { if (String(dataSiswa[i][0]) === String(catatanData.nis)) { targetRow = i + 1; saldoPoinSebelumnya = Number(dataSiswa[i][3]) || 100; namaSiswa = String(dataSiswa[i][1]); kelasSiswa = String(dataSiswa[i][2]); break; } } if (targetRow === -1) { throw new Error('Murid dengan NIS ' + catatanData.nis + ' tidak ditemukan di database.'); } const perubahanPoin = Number(catatanData.poin); const saldoAkhir = Math.max(0, saldoPoinSebelumnya + perubahanPoin); sheetSiswa.getRange(targetRow, 4).setValue(saldoAkhir); sheetSiswa.getRange(targetRow, 9).setValue(new Date()); const recordId = 'REC-' + new Date().getTime(); const now = new Date(); sheetRiwayat.appendRow([ recordId, now, catatanData.nis, namaSiswa, kelasSiswa, catatanData.tipe, catatanData.judul, perubahanPoin, saldoAkhir, catatanData.tindakan || '-', catatanData.guru || 'Tim Kesiswaan' ]); } } /** * Mendaftarkan murid baru secara perorangan */ function tambahSantriBaru(santriData) { CacheService.getScriptCache().remove('initial_data_azzahra_v2'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('DataSiswa'); if (!sheet) throw new Error('Sheet DataSiswa tidak ditemukan.'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(santriData.nis)) { return { status: 'error', message: 'NIS ' + santriData.nis + ' sudah terdaftar.' }; } } sheet.appendRow([ String(santriData.nis), String(santriData.nama), String(santriData.kelas), 100, String(santriData.wali || ''), String(santriData.hpWali || ''), 'Aktif', String(santriData.catatan || '-'), new Date(), String(santriData.waliKelas || ''), String(santriData.hpWaliKelas || '') ]); return { status: 'success', message: 'Murid baru berhasil ditambahkan.' }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Mendaftarkan murid secara massal dari impor berkas Excel */ function tambahSantriBatch(santriList) { CacheService.getScriptCache().remove('initial_data_azzahra'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('DataSiswa'); if (!sheet) throw new Error('Sheet DataSiswa tidak ditemukan.'); const data = sheet.getDataRange().getValues(); const existingNisSet = new Set(); for (let i = 1; i < data.length; i++) { if (data[i][0]) existingNisSet.add(String(data[i][0]).trim()); } const now = new Date(); const rowsToAppend = []; let duplikatCount = 0; santriList.forEach(s => { const nisClean = String(s.nis).trim(); if (!nisClean) return; if (existingNisSet.has(nisClean)) { duplikatCount++; return; } existingNisSet.add(nisClean); rowsToAppend.push([ nisClean, String(s.nama).trim(), String(s.kelas || '1 Umar bin Abdul Aziz').trim(), Number(s.poin) || 100, String(s.wali || ''), String(s.hpWali || ''), 'Aktif', String(s.catatan || '-'), now, String(s.waliKelas || ''), String(s.hpWaliKelas || '') ]); }); if (rowsToAppend.length > 0) { const startRow = sheet.getLastRow() + 1; sheet.getRange(startRow, 1, rowsToAppend.length, 11).setValues(rowsToAppend); } return { status: 'success', insertedCount: rowsToAppend.length, duplikatCount: duplikatCount, message: 'Alhamdulillah, ' + rowsToAppend.length + ' murid berhasil diimpor ke Google Sheets' + (duplikatCount > 0 ? ' (' + duplikatCount + ' murid dilewati karena NIS sudah ada).' : '.') }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Memperbarui status individual murid (Aktif, Lulus, Mutasi) */ function updateStatusSantri(nis, statusData) { CacheService.getScriptCache().remove('initial_data_azzahra'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('DataSiswa'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(nis)) { if (statusData.status) sheet.getRange(i + 1, 7).setValue(statusData.status); if (statusData.kelas) sheet.getRange(i + 1, 3).setValue(statusData.kelas); if (typeof statusData.poin !== 'undefined') sheet.getRange(i + 1, 4).setValue(statusData.poin); sheet.getRange(i + 1, 9).setValue(new Date()); return { status: 'success', message: 'Status murid berhasil diperbarui.' }; } } return { status: 'error', message: 'NIS tidak ditemukan.' }; } catch (error) { return { status: 'error', message: error.toString() }; } } /** * Pembaruan massal kelas, status, dan informasi wali kelas baru saat kenaikan kelas */ function batchUpdateSiswa(updatesList) { CacheService.getScriptCache().remove('initial_data_azzahra'); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('DataSiswa'); if (!sheet) throw new Error('Sheet DataSiswa tidak ditemukan.'); const data = sheet.getDataRange().getValues(); const now = new Date(); const lookupMap = {}; updatesList.forEach(u => { lookupMap[String(u.nis)] = u; }); let updatedCount = 0; for (let i = 1; i < data.length; i++) { const rowNis = String(data[i][0]); if (lookupMap[rowNis]) { const item = lookupMap[rowNis]; if (item.kelas) sheet.getRange(i + 1, 3).setValue(item.kelas); if (typeof item.poin !== 'undefined') sheet.getRange(i + 1, 4).setValue(item.poin); if (item.status) sheet.getRange(i + 1, 7).setValue(item.status); if (item.waliKelas) sheet.getRange(i + 1, 10).setValue(item.waliKelas); if (item.hpWaliKelas) sheet.getRange(i + 1, 11).setValue(item.hpWaliKelas); sheet.getRange(i + 1, 9).setValue(now); updatedCount++; } } return { status: 'success', message: 'Alhamdulillah, ' + updatedCount + ' data murid berhasil diperbarui di Google Sheets.' }; } catch (error) { return { status: 'error', message: error.toString() }; } }
